  var Fault = class extends Error {
    constructor(code, message, detail = {}) {
      super(message);
      this.code = code;
      this.detail = detail;
    }
  };
  var fail = (code, message, detail) => {
    throw new Fault(code, message, detail);
  };
  var norm = (v) => String(v ?? "").normalize("NFD").replace(/[\u0300-\u036f]/g, "").replace(/[đĐ]/g, "d").toLowerCase().replace(/\s+/g, " ").trim();
  var present = (v) => v !== void 0 && v !== null && String(v).trim() !== "";
  var uid = () => globalThis.crypto.randomUUID();
  function check(signal) {
    if (signal?.aborted) throw signal.reason || new Fault("STOPPED", "Đã dừng.");
  }
  function delay(ms, signal) {
    check(signal);
    return new Promise((resolve2, reject) => {
      let timer;
      const done = () => {
        clearTimeout(timer);
        signal?.removeEventListener("abort", abort);
        resolve2();
      };
      const abort = () => {
        clearTimeout(timer);
        signal?.removeEventListener("abort", abort);
        reject(signal.reason || new Fault("STOPPED", "Đã dừng."));
      };
      timer = setTimeout(done, ms);
      signal?.addEventListener("abort", abort, { once: true });
    });
  }
  async function wait(fn, { timeout = 2e4, interval = 100, signal, label = "giao diện" } = {}) {
    const end = Date.now() + timeout;
    while (Date.now() < end) {
      check(signal);
      const value = await fn();
      if (value) return value;
      await delay(Math.min(interval, end - Date.now()), signal);
    }
    fail("TIMEOUT", "Hết thời gian chờ " + label);
  }
  async function retryRead(fn, { attempts = 3, signal, onAttempt = () => {
  } } = {}) {
    for (let n = 1; n <= attempts; n++) {
      check(signal);
      onAttempt(n);
      try {
        return await fn();
      } catch (e) {
        if (!["TIMEOUT", "NETWORK"].includes(e.code) || n === attempts) throw e;
        await delay(250 * 2 ** (n - 1), signal);
      }
    }
  }
  function date(raw, { order = "DMY", excel = false } = {}) {
    if (!present(raw)) fail("DATE", "Thiếu ngày.");
    let y, m, d;
    if (excel && typeof raw === "number") {
      if (!Number.isInteger(raw) || raw < 1 || raw === 60)
        fail("DATE", "Excel serial không hợp lệ.");
      const t2 = new Date(
        Date.UTC(1899, 11, 31) + (raw - (raw > 60 ? 1 : 0)) * 864e5
      );
      [y, m, d] = [t2.getUTCFullYear(), t2.getUTCMonth() + 1, t2.getUTCDate()];
    } else {
      const s = String(raw).trim();
      let x = s.match(
        /^(\d{4})[-/.](\d{1,2})[-/.](\d{1,2})(?:T00:00:00(?:\.000)?Z?)?$/
      );
      if (x) [y, m, d] = x.slice(1).map(Number);
      else {
        x = s.match(/^(\d{1,2})[-/.](\d{1,2})[-/.](\d{4})$/);
        if (!x) fail("DATE", "Ngày phải là dd/mm/yyyy hoặc yyyy-mm-dd.");
        const [a, b, c] = x.slice(1).map(Number);
        if (order === "AUTO") {
          if (a <= 12 && b <= 12 && a !== b)
            fail("DATE_AMBIGUOUS", "Ngày mơ hồ; chọn DMY/MDY một lần cho nguồn.");
          if (a > 12 && b > 12) fail("DATE", "Ngày không hợp lệ.");
          [y, m, d] = [c, a > 12 ? b : a, a > 12 ? a : b];
        } else [y, m, d] = order === "MDY" ? [c, a, b] : [c, b, a];
      }
    }
    const t = new Date(Date.UTC(y, m - 1, d));
    if (y < 1900 || y > 2100 || t.getUTCFullYear() !== y || t.getUTCMonth() + 1 !== m || t.getUTCDate() !== d)
      fail("DATE", "Ngày không hợp lệ.");
    return `${y}-${String(m).padStart(2, "0")}-${String(d).padStart(2, "0")}`;
  }

          row = [];

      }

      row.push(field2);
      rows.push(row);

      contains: inside,
      querySelectorAll: (selector) => [...root.querySelectorAll(selector)].filter(inside)
    };
  }
  async function writeDate(find, raw, { guard, signal } = {}) {
    const wanted = String(raw ?? "").trim() ? date(raw) : "";
    guard();
    check(signal);
    const getInstance = (el) => {
      const host = el.closest(".dx-datebox,dx-date-box");
      if (!host) return null;
      const w2 = el.ownerDocument.defaultView;
      try {
        return w2.DevExpress?.ui?.dxDateBox?.getInstance(host) || w2.jQuery?.(host)?.dxDateBox?.("instance") || null;
      } catch {
        return null;
      }
    };
    const instance = getInstance(find());
    if (!instance)
      return write(
        find,
        !wanted ? "" : find().type === "date" ? wanted : vnDate(wanted),
        {
          guard,
          signal,
          column: "date"
        }
      );
    if (instance.option("readOnly") || instance.option("disabled"))
      fail("WIDGET", "Ngày khám bị khóa.");
    const [y, m, d] = wanted.split("-").map(Number);
    const w = find().ownerDocument.defaultView;
    instance.option("value", wanted ? new w.Date(y, m - 1, d) : null);
    await wait(
      () => {
        guard();
        const value = getInstance(find())?.option("value");
        if (!wanted) return value == null || value === "";
        if (value && typeof value.getFullYear === "function")
          return [
            value.getFullYear(),
            String(value.getMonth() + 1).padStart(2, "0"),
            String(value.getDate()).padStart(2, "0")
          ].join("-") === wanted;
        try {
          return date(value) === wanted;
        } catch {
          return false;
        }
      },
      { signal, timeout: 5e3, label: "ngày khám trong model datebox" }
    );
  }

  // src/rules.js
  function findingsFor(values, model) {
    setRulesModel(model);
    const normalized = { ...values }, issues = [];
    for (const f of CAN_LAM_SANG_FIELD_MAP) {
      const raw = values[f.column];
      if (!present(raw)) {
        if (f.column !== "Ure")
          issues.push({
            column: f.column,
            label: f.label,
            value: "",
            level: "missing",
            note: "Chưa có chỉ số này trong nguồn; không coi là bình thường."
          });
        continue;
      }
      try {
        const unit = CLS_ALERT_CONFIG[f.column]?.unit || "";
        const n = normalizedLabValue(f.column, raw, {
          sourceUnit: values[f.column + "_UNIT"] || "",
          targetUnit: unit
        });
        normalized[f.column] = n.write;
        if (values[f.column + "_UNIT"]) normalized[f.column + "_UNIT"] = unit;
      } catch {
        issues.push({
          column: f.column,
          label: f.label,
          value: String(raw),
          level: "review",
          note: "Cần xác minh giá trị, đơn vị hoặc mức định tính nguồn."
        });
        delete normalized[f.column];
      }
    }
    if (!present(values.NIT))
      issues.push({
        column: "NIT",
        label: "Nitrit",
        value: "",
        level: "missing",
        note: "Chưa có Nitrit trong nguồn."
      });
    return [...checkAbnormalResults(normalized), ...issues];
  }
  function proposal({
    model,
    health,
    disease,
    values = null,
    noLab = false
  }) {
    if (!health || !/^loai (i|ii|iii|iv|v)$/.test(norm(health)))
      fail(
        "DATA_REVIEW",
        "Phân loại sức khỏe còn trống/chưa đọc được; không chọn Khác trống và không lưu."
      );
    if (disease === null || !String(disease).trim())
      fail(
        "DATA_REVIEW",
        "Chưa đọc được bệnh/tật; không xem dữ liệu trống là bình thường."
      );
    const classification = ["M3", "M4"].includes(model) ? classifyBenhTatM34(disease) : classifyBenhTat(disease);
    if (["unknown", "stop"].includes(classification.type))
      fail("DATA_REVIEW", "Mã ICD cần kiểm tra theo quy tắc của mẫu.", {
        classification: classification.type
      });
    const findings = values && !noLab ? findingsFor(values, model) : [];
    if (findings.some((f) => ["review", "urgent", "missing"].includes(f.level)))
      fail(
        "DATA_REVIEW",
        "CLS có cảnh báo cần bác sĩ đánh giá hoặc thiếu bối cảnh."
      );
    const labRouting = analyzeReferralDepartments(
      "",
      findings.map((f) => ({ ...f, key: f.column }))
    );
    const icdParts = classification.type === "specialty" ? classification.text.split(", ") : [];
    const labParts = labRouting.departments;
    const aliases = {
      "Khám Răng": "Răng Hàm Mặt",
      "Khám Mắt": "Mắt",
      "Khám Tai Mũi Họng": "Tai Mũi Họng",
      "Khám Tim mạch": "Nội tim mạch",
      "Khám Nội tiết": "Nội tiết",
      "Khám Da liễu": "Da liễu",
      "Khám Cơ Xương Khớp": "Nội Thần kinh Cơ xương khớp",
      "Khám Thần kinh": "Nội Thần kinh Cơ xương khớp"
    };
    const taken = new Set(icdParts.map((s) => aliases[s] || s));
    const extras = labParts.filter((s) => !taken.has(s));
    let text = icdParts.join(", ");
    if (extras.length)
      text = [
        text,
        buildReferralProposal(
          buildCombinedReferralObservation("", findings),
          extras
        )
      ].filter(Boolean).join("\n");
    if (icdParts.length && !extras.length)
      text += " tại Bệnh viện Đa khoa khu vực Hóc Môn để được khám và tư vấn chi tiết hơn.";
    if (noLab)
      text = [text, "Công dân chưa có kết quả xét nghiệm"].filter(Boolean).join("\n");
    const normal = !text && classification.type === "normal";
    return {
      selection: normal ? "Bình thường, hẹn khám định kỳ lần sau" : "Khác",
      text: normal && model === "M11" ? "Bình thường, hẹn khám định kỳ lần sau" : text,
      findings,
      classification: classification.type,
      noLab
    };
  }
  function combineExisting(existing, incoming, mode = "keep") {
    const a = String(existing ?? "").trim(), b = String(incoming ?? "").trim();
    if (!a || norm(a) === norm(b)) return b || a;
    if (mode === "replace") return b;
    if (mode === "append") {
      const lines = a.split("\n");
      return [
        ...lines,
        ...b.split("\n").filter((s) => !lines.some((l) => norm(l) === norm(s)))
      ].join("\n");
    }
    fail(
      "EXISTING_DATA",
      "Nội dung đã có khác đề xuất. Chọn Bổ sung/Thay thế có chủ đích hoặc giữ nguyên và xử lý tay."
    );
  }

  function captureFilters(root) {
    const inputs = controls(root);
    return inputs.flatMap((el) => {
      let label = all(root, "label").find((l) => l.htmlFor && l.htmlFor === el.id)?.textContent?.trim();
      if (!label) {
        for (let p = el.parentElement, depth = 0; p && root.contains(p) && depth < 4; p = p.parentElement, depth++) {
          if (inputs.filter((i) => p.contains(i)).length !== 1) break;
          const labels = all(p, "label,.control-label,.dx-field-item-label-text");
          if (labels.length === 1) {
            label = labels[0].textContent.trim();
            break;
          }
        }
      }
      const key = aliasKey(
        label || el.getAttribute("aria-label") || el.name || ""
      );
      const locator = label ? { names: [label] } : el.name ? { name: el.name } : el.getAttribute("aria-label") ? { names: [el.getAttribute("aria-label")] } : el.id ? { id: el.id } : null;
      if (!locator) {
        if (read(el))
          fail(
            "UI_CHANGED",
            "Bộ lọc có giá trị nhưng chưa có nhãn/định danh để khôi phục."
          );
        return [];
      }
      return [
        {
          ...locator,
          key,
          value: read(el),
          isDate: el.type === "date" || !!el.closest(".dx-datebox,dx-date-box") || /ngày|ngay/i.test(label || el.name || el.id)
        }
      ];
    });
  }
  async function restoreFilters(adapter, filters) {
    let changed = false;
    const guard = () => {
      if (adapter.context(false).page !== "list")
        fail("CONTEXT", "Đã rời danh sách.");
    };
    for (const f of filters || []) {
      const find = () => {
        const root = adapter.context(false).root;
        if (f.names || !f.id && !f.name && f.key)
          return field(root, f.names || ALIASES[f.key], { forRead: true });

          controls(root).filter(
            (el) => f.name ? el.name === f.name : el.id === f.id
          ),
          "Bộ lọc danh sách"
        );
      };
      if (read(find()) === f.value) continue;
      changed = true;
      await (f.isDate ? writeDate : write)(find, f.value, {
        guard,
        signal: adapter.signal
      });
    }

  }

  function questionnaire(root) {
    const candidates = all(
      root,
      '.dx-list,.dx-radiogroup,fieldset,[role="radiogroup"],.dx-checkbox,input[type="checkbox"]'
    );
    const groups = candidates.filter(
      (g) => !candidates.some((x) => x !== g && g.contains(x))
    );
    const out = [];
    for (const group of groups) {
      let key = group.getAttribute("aria-label") || "";
      const id = group.getAttribute("aria-labelledby");
      if (id) key = group.ownerDocument.getElementById(id)?.textContent || key;
      if (!key)
        key = group.closest("fieldset")?.querySelector(":scope > legend")?.textContent || "";
      if (!key) {
        let p = group.parentElement;
        for (let n = 0; n < 4 && p && root.contains(p); n++, p = p.parentElement) {
          const labels = all(p, "label,legend,b,strong").filter(
            (el) => !group.contains(el)
          );
          if (labels.length === 1) {
            key = labels[0].textContent;
            break;
          }
          if (labels.length > 1) break;
        }
      }
      const isCheckbox = group.matches(".dx-checkbox,input[type=checkbox]");
      if (isCheckbox) {
        const label = group.getAttribute("aria-label") || group.closest(".dx-checkbox")?.textContent || group.closest("label")?.textContent || group.textContent;
        const legend = group.closest("fieldset")?.querySelector("legend")?.textContent || key;
        if (!label?.trim())
          fail("UI_CHANGED", "Checkbox không có nhãn xác minh được.");
        key = legend + " | " + label;
      }
      if (!key.trim()) fail("UI_CHANGED", "Câu hỏi không có nhãn duy nhất.");
      const opts = [
        ...isCheckbox ? [group] : [],
        ...all(
          group,
          '.dx-list-item,.dx-radiobutton,[role="radio"],input[type="radio"],input[type="checkbox"],.dx-checkbox'
        )
      ];
      const leaves = opts.filter(
        (n) => !opts.some((x) => x !== n && n.contains(x))
      );
      const choices = leaves.map((el) => ({
        value: el.getAttribute("aria-label") || el.closest("label")?.textContent || el.textContent || el.parentElement.textContent,
        node: el.closest(".dx-list-item,.dx-radiobutton,.dx-checkbox") || el
      }));
      out.push({ key: norm(key), choices });
    }
    if (new Set(out.map((g) => g.key)).size !== out.length)
      fail("UI_CHANGED", "Nhãn câu hỏi bị trùng.");
    return out;
  }

  async function fillClinical(adapter, { mode }) {
    const get = () => questionnaire(adapter.guard("clinical").root);

          get().filter((g2) => g2.key === p.key),

        );

          g.choices.filter((c) => norm(c.value) === norm(p.value)),

        );

          fail("EXISTING_DATA", "Câu hỏi lâm sàng đã có đáp án khác.");

      fail("UI_CHANGED", "Chưa có nhóm Chưa phát hiện bất thường.");

      async () => {
        const groups = get();
        classes = actions(groups, "Loại I");
        if (!classes.length) return false;
        const sig = JSON.stringify(classes.map((p) => p.key).sort());
        if (sig !== signature) {
          signature = sig;
          since = 0;
        }
        for (const p of classes) {
          const g = groups.find((g2) => g2.key === p.key);
          const target = unique(
            g.choices.filter((c) => norm(c.value) === "loai i"),
            "Loại I"
          );
          if (!selected(target.node)) {
            await apply(p);
            since = 0;
            return false;
          }
        }
        if (!since) since = Date.now();
        return Date.now() - since >= 400;
      },
      {
        signal: adapter.signal,
        timeout: 2e4,
        label: "mọi Loại I đã render và giữ lựa chọn"
      }
    );

        final.filter((g) => g.key === p.key),

    }
    return {

      fields: normal.length + classes.length + dropdownCount + refusal.length
    };

  var STORAGE_KEY = "tro_ly_medinet_v1";
  var Store = class {
    constructor(storage = globalThis.sessionStorage) {
      this.storage = storage;
      this.data = { schema: 1, checkpoint: null, logs: [] };
      const saved = storage.getItem(STORAGE_KEY);
      if (saved) {
        try {
          const parsed = JSON.parse(saved);
          if (parsed.schema === 1 && Array.isArray(parsed.logs))
            this.data = parsed;
          else fail("RECOVERY", "Checkpoint không tương thích.");
        } catch {
          fail("RECOVERY", "Không đọc được checkpoint. Không tự ghi tiếp.");
        }
      }
    }
    save() {
      try {
        this.storage.setItem(STORAGE_KEY, JSON.stringify(this.data));
      } catch {
        fail(
          "STORAGE",
          "Không lưu được checkpoint; dừng trước bước ghi tiếp theo."
        );
      }
    }
    checkpoint(value) {
      if (arguments.length) {
        this.data.checkpoint = value;
        this.save();
      }
      return this.data.checkpoint;
    }
    log(entry) {
      this.data.logs.push(entry);
      this.save();
    }
    finalize(entry, key) {
      const cp = this.data.checkpoint;
      if (!cp.finished.includes(key)) {
        const exists = this.data.logs.some(
          (l) => l.runId === cp.runId && l.case && caseKey(l.case) === key && ["DONE", "NO_LAB_VERIFIED"].includes(l.status)
        );
        if (!exists) this.data.logs.push(entry);
        cp.finished.push(key);
        this.save();
      }
    }
    completed(c) {
      return this.data.logs.some(
        (l) => l.case && l.case.visitId === c.visitId && l.case.date === c.date && l.case.model === c.model && ["DONE", "NO_LAB_VERIFIED"].includes(l.status)
      );
    }
  };
  async function exclusive(locks, fn) {
    if (!locks?.request)
      fail(
        "LOCK",
        "Trình duyệt không hỗ trợ Web Locks; batch bị khóa để tránh xử lý chồng tab."
      );
    return locks.request(
      "tro-ly-medinet:one-runner-per-origin",
      { mode: "exclusive", ifAvailable: true },
      (lock) => {
        if (!lock) fail("LOCK", "Một tab khác đang chạy Trợ lý Medinet.");
        return fn();
      }
    );
  }

  function bodyIdentity(body, c) {

      const data = JSON.parse(body);

        if (depth > 12 || !v || typeof v !== "object") return;

            found = true;

      );

      this.el("progress").textContent = `Tổng lượt đã nhận diện: ${cp?.planned?.length || 0} · Bước: ${cp?.step || "Chưa chạy"} · Lượt: ${cp?.case?.visitId || "—"} · Hoàn tất: ${final.filter((l) => l.status === "DONE").length} · Thiếu XN đã xác minh: ${final.filter((l) => l.status === "NO_LAB_VERIFIED").length} · Cần kiểm tra: ${outcomes.filter((l) => l.status === "NEEDS_REVIEW").length} · Lỗi: ${outcomes.filter((l) => l.status === "TECHNICAL_ERROR").length}`;
      this.el("state").textContent = this.runner.running ? "Đang chạy" : cp?.error ? "Đã tạm dừng" : cp?.step === "COMPLETE" ? "Đã kết thúc" : "Chờ thao tác";
      this.el("source").textContent = this.labs.loadedAt ? "Nguồn CLS đã tải lúc: " + new Date(this.labs.loadedAt).toLocaleString("vi-VN") : "Nguồn CLS: chưa tải";
      if (cp?.error) this.message(cp.error.message);
    }
  };

  // src/main.js
  function boot(win = window) {
    if (win.top !== win || win.__troLyMedinet) return;
    const service = {};
    win.__troLyMedinet = service;
    const store = new Store(win.sessionStorage), labs = new LabData({ fetcher: win.fetch.bind(win) }), network = new NetworkTracker(win);
    network.install();
    let panel;
    const runner = new Runner({
      win,
      store,
      labs,
      network,
      onChange: () => panel?.refresh()
    });
    panel = new Panel({ win, runner, labs });
    Object.assign(service, { runner, panel, network, labs });
    let pending = null;
    const schedule = () => {
      clearTimeout(pending);
      pending = setTimeout(() => panel.refresh(), 150);
    };
    const push = win.history.pushState, replace = win.history.replaceState;
    win.history.pushState = function(...args) {
      const value = push.apply(this, args);
      schedule();
      return value;
    };
    win.history.replaceState = function(...args) {
      const value = replace.apply(this, args);
      schedule();
      return value;
    };
    win.addEventListener("popstate", schedule);
    const capture = (event) => {
      try {
        const c = resolve(win.document, win.location.href, {
          requireIdentity: false
        });
        if (c.page !== "list" || !principal(win)) return;
        const action = event.target.closest("a,button,[role=button]");
        if (!action) return;
        const match = listRows(c).find((r) => r.case && r.row.contains(action));
        if (!match) return;
        store.data.context = {
          origin: win.location.origin,
          owner: principal(win),
          case: match.case,
          updatedAt: Date.now()
        };
        store.save();
      } catch {
      }
    };
    win.document.addEventListener("click", capture, true);
    const observer = new win.MutationObserver(schedule);
    const content = win.document.querySelector('main,[role="main"],.content');
    const root = content?.parentElement || win.document.body;
    observer.observe(root, {
      subtree: true,
      childList: true,
      attributes: true,
      attributeFilter: ["hidden", "aria-hidden", "aria-busy"]
    });
    service.dispose = () => {
      observer.disconnect();
      win.document.removeEventListener("click", capture, true);
      clearTimeout(pending);
      win.removeEventListener("popstate", schedule);
      win.history.pushState = push;
      win.history.replaceState = replace;
      panel.dispose();
      network.dispose();
    };
    win.addEventListener("pagehide", service.dispose, { once: true });
    labs.bindCache = () => {
      const owner = principal(win);
      if (owner !== labs.cacheOwner) {
        labs.loadedAt = 0;
        labs.cache = owner ? new SourceCache(win.indexedDB, "DATA|" + owner) : null;
        labs.cacheOwner = owner;
      }
    };
    labs.bindCache();
    const cp = store.checkpoint();
    if (cp?.active && cp.auto)
      wait(() => principal(win), {
        timeout: 3e4,
        label: "phiên đăng nhập để kiểm tra checkpoint"
      }).then(() => runner.resume({ automatic: true })).catch((e) => panel.message(e.message));
    return service;
  }
  if (typeof window !== "undefined" && !globalThis.__MEDINET_TEST__) boot();
})();
